function galBuild(key){
  const sec=galSec[key];const tabs=$(key+'Tabs');if(!tabs)return;

function galLoad(key,i){
  const sec=galSec[key];sec.idx=i;
  sec.items.forEach((_,j)=>{const t=$(key+'Tab'+j);if(t)t.classList.toggle('active',j===i)});
  const ex=sec.items[i];
  $(key+'Title').textContent=ex.icon+' '+ex.title;
  $(key+'Desc').textContent=ex.desc;
  $(key+'Q').textContent='❓ '+ex.q;
  $(key+'Opts').innerHTML=ex.opts.map(o=>`<button class="shp-gal-opt" onclick="galAnswer('${key}',this,'${o.replace(/'/g,"\\'")}')">${o}</button>`).join('');
  const rv=$(key+'Reveal');rv.style.display='none';rv.innerHTML='';
  setTimeout(()=>galDraw(key,ex.draw),60);
}
function galAnswer(key,btn,ans){
  const sec=galSec[key],ex=sec.items[sec.idx],ok=(ans===ex.ans);
  document.querySelectorAll('#'+key+'Opts .shp-gal-opt').forEach(b=>{
    b.disabled=true;
    const t=b.textContent.trim();
    if(t===ex.ans)b.classList.add('correct');
    else if(t===ans&&!ok)b.classList.add('wrong');
  });
  const rv=$(key+'Reveal');rv.style.display='block';
  rv.innerHTML=(ok?'🎉 정답! ':'💡 정답은 <strong>'+ex.ans+'</strong>!<br>')+ex.reveal;
  if(ok&&!sec.done[sec.idx]){
    sec.done[sec.idx]=true;
    const t=$(key+'Tab'+sec.idx);if(t)t.classList.add('done');
    boom('발견!','var(--teal)');
  }
}
