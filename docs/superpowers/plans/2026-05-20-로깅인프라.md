**Goal:** 학생이 코드+이름으로 로그인하고, microgenetic 분석용 trial 로그를 Google Sheets로 비동기 수집하며, 닮음비·명화 탭을 제거하고 소리/악기 탭을 분리한다. 색상 실험실에 등가비 task를 얹어 trial 로깅의 reference로 삼는다.

```javascript
// RatioSense 로그 수집 백엔드 (Google Apps Script Web App)
// 배포: 편집기 → 배포 → 새 배포 → 웹 앱 → 액세스: '모든 사용자'
// 스크립트 속성에 TOKEN 설정: 프로젝트 설정 → 스크립트 속성 → TOKEN = ratiosense-2026

var SHEETS = {
  sessions: ['schemaVersion','sessionId','userCode','userName','codeValid','startTs','endTs','userAgent','screen','lang','appVersion','seq','clientTs'],
  trials:   ['schemaVersion','trialId','sessionId','userCode','userName','tab','activityId','taskId','attemptNo','startTs','endTs','outcome','trajectory','finalState','triggerRef','seq','clientTs'],
  events:   ['schemaVersion','eventId','trialId','sessionId','userCode','userName','tab','eventType','payload','seq','clientTs']
};

function doGet() {
  return json_({ ok: true, service: 'ratiosense-log' });
}

function doPost(e) {
  try {
    var token = PropertiesService.getScriptProperties().getProperty('TOKEN');
    var body = JSON.parse(e.postData.contents);
    if (!token || body.token !== token) {
      return json_({ ok: true, accepted: 0 }); // 조용히 무시(봇 차단)
    }
    var recs = body.records || [];
    var byType = { sessions: [], trials: [], events: [] };
    for (var i = 0; i < recs.length; i++) {
      var r = recs[i];
      var t = r._t; // 'sessions' | 'trials' | 'events'
      if (byType[t]) {
        if (t === 'sessions') r.codeValid = isValidCode_(r.userCode);
        byType[t].push(r);
      }
    }
    var accepted = 0;
    for (var t2 in byType) {
      if (byType[t2].length) accepted += appendRows_(t2, byType[t2]);
    }
    return json_({ ok: true, accepted: accepted });
  } catch (err) {
    return json_({ ok: false, error: String(err) });
  }
}

function appendRows_(sheetName, rows) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
  var header = SHEETS[sheetName];
  if (sh.getLastRow() === 0) sh.appendRow(header);
  var matrix = rows.map(function (r) {
    return header.map(function (h) {
      var v = r[h];
      if (v && typeof v === 'object') return JSON.stringify(v);
      return v == null ? '' : v;
    });
  });
  sh.getRange(sh.getLastRow() + 1, 1, matrix.length, header.length).setValues(matrix);
  return matrix.length;
}

function isValidCode_(code) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName('roster');
  if (!sh || sh.getLastRow() < 2) return '';
  var codes = sh.getRange(2, 1, sh.getLastRow() - 1, 1).getValues().map(function (r) { return String(r[0]).trim(); });
  return codes.indexOf(String(code).trim()) >= 0;
}

function json_(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
```

```javascript
/* ===== RatioSense Logger (microgenetic trial logging) ===== */
const LOG_CFG = {
  endpoint: '',           // 배포 후 Apps Script 웹 앱 URL로 교체
  token: 'ratiosense-2026',
  appVersion: 'dev',
  batchSize: 10, batchMs: 5000, debounceMs: 400,
  maxQueue: 500, lsKey: 'rs_logq', sessKey: 'rs_sess'
};
const Logger = (function () {
  //__LOGGER_PURE_START__
  function decideBatch(queueLen, msSinceLast, cfg) {
    if (queueLen <= 0) return false;
    return queueLen >= cfg.batchSize || msSinceLast >= cfg.batchMs;
  }
  function applyOverflow(queue, max) {
    let dropped = 0;
    while (queue.length > max) { queue.shift(); dropped++; }
    return dropped;
  }
  function coalesce(points) {
    const out = [];
    for (const p of points) {
      const last = out[out.length - 1];
      if (last && JSON.stringify(last.state) === JSON.stringify(p.state)) out[out.length - 1] = p;
      else out.push(p);
    }
    return out;
  }
  //__LOGGER_PURE_END__

  let queue = [], seq = 0, sessionId = null, user = { code: '', name: '' };
  let lastSend = Date.now(), sending = false, droppedTotal = 0;
  const trials = {}; // trialId -> {points, debTimer, lastState}

  function safe(fn) { try { return fn(); } catch (e) { /* 로깅은 절대 앱을 멈추지 않음 */ } }
  function uuid() { return 'x' + Date.now().toString(36) + Math.random().toString(36).slice(2, 8); }
  function backup() { safe(() => localStorage.setItem(LOG_CFG.lsKey, JSON.stringify(queue))); }
  function restore() { safe(() => { const s = localStorage.getItem(LOG_CFG.lsKey); if (s) queue = JSON.parse(s) || []; }); }

  function enqueue(rec) {
    safe(() => {
      rec.schemaVersion = '1';
      rec.sessionId = sessionId;
      rec.userCode = user.code;
      rec.userName = user.name;
      rec.seq = ++seq;
      rec.clientTs = new Date().toISOString();
      queue.push(rec);
      droppedTotal += applyOverflow(queue, LOG_CFG.maxQueue);
      backup();
      if (decideBatch(queue.length, Date.now() - lastSend, LOG_CFG)) flush();
    });
  }

  return { _pure: { decideBatch, applyOverflow, coalesce }, _q: () => queue, enqueue, _uuid: uuid };
})();
```

- [ ] **Step 1: Logger IIFE 안 `return` 직전에 전송 함수 추가**

```javascript
  let retryDelay = 1000;
  function flush(useBeacon) {
    safe(() => {
      if (!queue.length || !LOG_CFG.endpoint) return;
      if (sending && !useBeacon) return;
      const batch = queue.slice(0);
      const body = JSON.stringify({ token: LOG_CFG.token, records: batch });
      lastSend = Date.now();
      if (useBeacon && navigator.sendBeacon) {
        const ok = navigator.sendBeacon(LOG_CFG.endpoint, new Blob([body], { type: 'text/plain' }));
        if (ok) { queue = []; backup(); }
        return;
      }
      sending = true;
      fetch(LOG_CFG.endpoint, { method: 'POST', headers: { 'Content-Type': 'text/plain' }, body })
        .then(r => r.json())
        .then(() => { queue.splice(0, batch.length); backup(); sending = false; retryDelay = 1000; })
        .catch(() => {
          sending = false;
          retryDelay = Math.min(retryDelay * 2, 30000);
          setTimeout(() => flush(), retryDelay); // 큐는 그대로 → 재전송
        });
    });
  }
  setInterval(() => flush(), LOG_CFG.batchMs);
  document.addEventListener('visibilitychange', () => { if (document.visibilityState === 'hidden') flush(true); });
  window.addEventListener('pagehide', () => flush(true));
```

```javascript
  function startTrial(o) {
    const trialId = uuid();
    trials[trialId] = { points: [], debTimer: null, lastState: null };
    enqueue({ _t: 'events', eventId: uuid(), trialId, tab: o.tab, eventType: 'trial_start',
      payload: JSON.stringify({ activityId: o.activity, taskId: o.task, attemptNo: o.attemptNo || 1 }) });
    trials[trialId].meta = { tab: o.tab, activity: o.activity, task: o.task, attemptNo: o.attemptNo || 1, startTs: new Date().toISOString() };
    return trialId;
  }
  function trialState(trialId, state) {
    const tr = trials[trialId]; if (!tr) return;
    if (tr.debTimer) clearTimeout(tr.debTimer);
    tr.debTimer = setTimeout(() => safe(() => {
      tr.points.push({ t: Date.now(), state });
      tr.points = coalesce(tr.points);
      enqueue({ _t: 'events', eventId: uuid(), trialId, tab: tr.meta.tab, eventType: 'trial_state', payload: JSON.stringify(state) });
    }), LOG_CFG.debounceMs);
  }
  function endTrial(trialId, o) {
    const tr = trials[trialId]; if (!tr) return;
    if (tr.debTimer) clearTimeout(tr.debTimer);
    enqueue({ _t: 'trials', trialId, tab: tr.meta.tab, activityId: tr.meta.activity, taskId: tr.meta.task,
      attemptNo: tr.meta.attemptNo, startTs: tr.meta.startTs, endTs: new Date().toISOString(),
      outcome: o.outcome, trajectory: tr.points, finalState: o.finalState || null, triggerRef: o.triggerRef || '' });
    delete trials[trialId];
  }
```
