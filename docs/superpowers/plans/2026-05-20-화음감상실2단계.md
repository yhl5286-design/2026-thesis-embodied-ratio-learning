function rateRatio(a,b){
  const m=Math.max(a,b);
  if(m<=2) return {stars:5,level:'완벽한 어울림'};
  if((a===2&&b===3)||(a===3&&b===2)) return {stars:5,level:'아주 잘 어울림'};
  if(m<=6) return {stars:4,level:'잘 어울림'};
  if(m<=10) return {stars:3,level:'보통'};
  if(m<=16) return {stars:2,level:'약한 어울림'};
  return {stars:1,level:'잘 안 어울림'};
}
function reduceRatio(p,q){const g=gcd(p,q)||1;return [p/g,q/g];}
