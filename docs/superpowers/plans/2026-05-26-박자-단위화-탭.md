**Goal:** `unit`(단위화)·`beat`(박자) 두 신규 탭을 단일 `index.html`에 추가. 단위화는 한 마디(2.4s)를 음표 단위로 채우며 단위↔개수 역관계 발견. 박자는 두 메트로놈(좌 자동 슬라이더 + 우 학습자 두드리기) 폴리리듬에서 "같이 치는 순간" 발견. 각 탭 3 stage(만남/분리→박수선↔시간선→변환·불변량). microgenetic 로깅·W4 교량·W5 점도식 인프라 재사용.

```javascript
// ━━━━━━ TapExtractor — 두드림 → 박 수 추출 (±25% 관대) ━━━━━━
//__TAP_EXTRACTOR_START__
function medianInterval(times){
  if(!times||times.length<2)return 0;
  const gaps=[];
  for(let i=1;i<times.length;i++)gaps.push(times[i]-times[i-1]);
  gaps.sort((a,b)=>a-b);
  const mid=Math.floor(gaps.length/2);
  return gaps.length%2?gaps[mid]:(gaps[mid-1]+gaps[mid])/2;
}
function extractBeats(times,barMs){
  if(!times||times.length<2||barMs<=0)return 0;
  // 노이즈 합치기: 50ms 이내 두드림은 1박
  const cleaned=[times[0]];
  for(let i=1;i<times.length;i++){
    if(times[i]-cleaned[cleaned.length-1]>=50)cleaned.push(times[i]);
  }
  if(cleaned.length<2)return 0;
  const med=medianInterval(cleaned);
  if(med<=0)return 0;
  // 마디를 중앙 간격으로 나눈 추정 박 수, 관대도 ±25%
  const est=barMs/med;
  const rounded=Math.round(est);
  if(rounded<1)return 0;
  // 추정치가 정수와 25% 이상 벗어나면 실제 두드림 수 사용
  if(Math.abs(est-rounded)/rounded>0.25)return cleaned.length;
  return rounded;
}
//__TAP_EXTRACTOR_END__
```

```javascript
const UNIT_COUNTS={half:2,quarter:4,eighth:8,sixteenth:16,triplet:3};
function equivScale(ratio, k){
  const a=ratio[0]*k, b=ratio[1]*k;
  if(!Number.isInteger(a)||!Number.isInteger(b))return null;
  return [a,b];
}
function countForUnit(unit){
  return UNIT_COUNTS[unit]||0;
}
function unitFraction(unit){
  const c=UNIT_COUNTS[unit];
  return c?[1,c]:null;
}
function unitChange(from, to){
  const a=UNIT_COUNTS[from], b=UNIT_COUNTS[to];
  if(!a||!b)return {countMultiplier:0};
  return {countMultiplier:b/a};
}
//__RS_TRANSFORMS_END__
```

```javascript
// ━━━━━━ Metronome — Web Audio lookahead scheduler (박자·단위화 공용) ━━━━━━
const Metronome=(function(){
  let ctx=null;
  const tracks={}; // id -> {beats, barMs, freq, gain, onBeat, nextTime, lookahead, intervalId, lastBeatIdx}
  function getCtx(){try{if(!ctx)ctx=new (window.AudioContext||window.webkitAudioContext)();return ctx;}catch(e){return null;}}
  function playClick(time, freq, gain){
    const c=getCtx();if(!c)return;
    try{
      const osc=c.createOscillator();
      const g=c.createGain();
      osc.frequency.value=freq;
      osc.type='sine';
      g.gain.setValueAtTime(0,time);
      g.gain.linearRampToValueAtTime(gain,time+0.005);
      g.gain.exponentialRampToValueAtTime(0.0001,time+0.12);
      osc.connect(g).connect(c.destination);
      osc.start(time);
      osc.stop(time+0.15);
    }catch(e){}
  }
  function tick(id){
    const t=tracks[id];if(!t)return;
    const c=getCtx();if(!c)return;
    while(t.nextTime < c.currentTime + 0.1){
      // 한 박 재생
      const beatIdx=t.lastBeatIdx;
      playClick(t.nextTime, t.freq, t.gain);
      if(typeof t.onBeat==='function')try{t.onBeat(beatIdx, t.nextTime);}catch(e){}
      t.lastBeatIdx=(beatIdx+1)%t.beats;
      t.nextTime += t.barMs/t.beats/1000;
    }
  }
  function start(id, opts){
    try{
      const c=getCtx();if(!c)return;
      if(c.state==='suspended')c.resume();
      stop(id);
      const t={
        beats: opts.beats||2,
        barMs: opts.barMs||2400,
        freq: opts.freq||440,
        gain: opts.gain||0.15,
        onBeat: opts.onBeat||null,
        nextTime: c.currentTime+0.05,
        lastBeatIdx:0
      };
      tracks[id]=t;
      t.intervalId=setInterval(()=>tick(id), 25);
    }catch(e){}
  }
  function stop(id){
    try{
      const t=tracks[id];if(!t)return;
      if(t.intervalId)clearInterval(t.intervalId);
      delete tracks[id];
    }catch(e){}
  }
  function stopAll(){
    try{Object.keys(tracks).forEach(stop);}catch(e){}
  }
  function update(id, opts){
    const t=tracks[id];if(!t)return;
    if(typeof opts.beats==='number')t.beats=opts.beats;
    if(typeof opts.barMs==='number')t.barMs=opts.barMs;
    if(typeof opts.freq==='number')t.freq=opts.freq;
    if(typeof opts.gain==='number')t.gain=opts.gain;
    if(typeof opts.onBeat==='function')t.onBeat=opts.onBeat;
  }
  function isRunning(id){return !!tracks[id];}
  return {start, stop, stopAll, update, isRunning};
})();
```
